cmake_minimum_required(VERSION 3.19)

set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")

project(AnimalCrossingVita C CXX)

include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

# Project information
set(VITA_APP_NAME "Animal Crossing")
set(VITA_TITLEID  "ACVT00001")
set(VITA_VERSION  "02.00")

# Build configuration options
option(AC_INCLUDE_DECOMP "Include AC-Decomp game logic" OFF)
option(AC_ENABLE_DEBUG "Enable debug features" OFF)

# Build type configuration
set(AC_BUILD_TYPE "RUNTIME" CACHE STRING "Build type: RUNTIME, FULL_GAME, DEVELOPMENT")
set_property(CACHE AC_BUILD_TYPE PROPERTY STRINGS "RUNTIME" "FULL_GAME" "DEVELOPMENT")

# Compiler flags
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=gnu11 -Wall -O3 -mfloat-abi=hard")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++17 -Wall -O3 -mfloat-abi=hard")

# Debug configuration
if(AC_ENABLE_DEBUG)
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g -DAC_DEBUG=1")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -DAC_DEBUG=1")
endif()

# Include directories
include_directories(
    platform
    platform/gx_wrapper
)

# Base source files (always included)
set(BASE_SOURCES
    platform/ac_runtime_asset_loader.c
)

# Build type specific sources
if(AC_BUILD_TYPE STREQUAL "RUNTIME")
    message(STATUS "Building ENHANCED TEXTURE VPK - Adaptive rendering for GameCube formats")
    list(APPEND BASE_SOURCES 
         platform/vita_main_ultimate.c
         platform/enhanced_texture_renderer.c)
    
elseif(AC_BUILD_TYPE STREQUAL "GX_WRAPPER")
    message(STATUS "Building GX WRAPPER VPK - Comprehensive GameCube Format Support")
    list(APPEND BASE_SOURCES
         platform/vita_main_ultimate.c
         platform/ac_runtime_asset_loader.c
         platform/enhanced_texture_renderer.c)
    
elseif(AC_BUILD_TYPE STREQUAL "FULL_GAME")
    message(STATUS "Building complete AC-Decomp game")
    list(APPEND BASE_SOURCES 
        platform/vita_main_ac_game.c
        platform/gx_to_vitagl_wrapper.c
        platform/enhanced_texture_renderer.c
        platform/ac_runtime_asset_loader.c
    )
    
    # Include AC-Decomp sources if available and requested
    if(AC_INCLUDE_DECOMP AND EXISTS "${CMAKE_SOURCE_DIR}/../ac-decomp-upstream")
        message(STATUS "Including AC-Decomp game logic")
        include_directories("../ac-decomp-upstream/include")
        
        # Add AC-Decomp source files (selective inclusion)
        file(GLOB AC_DECOMP_SOURCES 
            "../ac-decomp-upstream/src/game/*.c"
            "../ac-decomp-upstream/src/actor/*.c"
            "../ac-decomp-upstream/src/effect/*.c"
        )
        list(APPEND BASE_SOURCES ${AC_DECOMP_SOURCES})
        
        # AC-Decomp specific flags
        add_definitions(-DAC_DECOMP_INTEGRATED=1)
    else()
        message(WARNING "AC-Decomp not found or not requested - using placeholder game logic")
        list(APPEND BASE_SOURCES platform/ac_game_placeholder.c)
    endif()
    
elseif(AC_BUILD_TYPE STREQUAL "DEVELOPMENT")
    message(STATUS "Building development version")
    list(APPEND BASE_SOURCES 
        platform/vita_main_dev.c
        platform/ac_debug_system.c
    )
    add_definitions(-DAC_DEV_BUILD=1)
    
else()
    message(FATAL_ERROR "Unknown build type: ${AC_BUILD_TYPE}")
endif()

# LiveArea configuration
set(VITA_MKSFOEX_FLAGS "${VITA_MKSFOEX_FLAGS} -d PARENTAL_LEVEL=1")

# Create executable
add_executable(AnimalCrossingVita ${BASE_SOURCES})

# Preprocessor definitions
target_compile_definitions(AnimalCrossingVita PRIVATE
    AC_VITA_PORT=1
    AC_BUILD_TYPE_${AC_BUILD_TYPE}=1
    AC_VITAGL_WRAPPER=1
    VITA_TITLEID="${VITA_TITLEID}"
)

# Auto-include VitaGL wrapper in all AC-decomp files for mass conversion
target_compile_options(AnimalCrossingVita PRIVATE
    -include "${CMAKE_SOURCE_DIR}/platform/gx_to_vitagl_wrapper.h"
)

# Link libraries (comprehensive set for all build types - based on working config)
target_link_libraries(AnimalCrossingVita
    # VitaGL stack (essential for graphics)
    vitaGL
    vitashark
    SceShaccCgExt
    mathneon
    
    # VitaGL required system stubs
    taihen_stub
    SceShaccCg_stub
    
    # Core Vita system libraries (essential for VitaGL)
    SceAppMgr_stub
    SceAppUtil_stub
    SceCtrl_stub
    SceDisplay_stub
    SceGxm_stub
    SceSysmodule_stub
    SceTouch_stub
    ScePower_stub
    
    # VitaGL graphics dependencies
    SceKernelDmacMgr_stub
    
    # Additional system support
    SceAudio_stub
    SceCommonDialog_stub
    SceHid_stub
    SceIofilemgr_stub
    ScePvf_stub
    
    # Standard libraries
    c
    m
)

# Create SELF and VPK (using working configuration from successful builds)
vita_create_self(AnimalCrossingVita.self AnimalCrossingVita UNCOMPRESSED)

# Manual VPK creation target (until we fix vita_create_vpk issues)
add_custom_target(vpk
    COMMAND ${CMAKE_COMMAND} -E echo "Creating VPK with LiveArea assets..."
    COMMAND vita-pack-vpk -s ${CMAKE_SOURCE_DIR}/sce_sys/param.sfo -b AnimalCrossingVita.self
        --add ${CMAKE_SOURCE_DIR}/sce_sys/icon0.png=sce_sys/icon0.png
        --add ${CMAKE_SOURCE_DIR}/sce_sys/livearea/contents/bg.png=sce_sys/livearea/contents/bg.png
        --add ${CMAKE_SOURCE_DIR}/sce_sys/livearea/contents/startup.png=sce_sys/livearea/contents/startup.png
        --add ${CMAKE_SOURCE_DIR}/sce_sys/livearea/contents/template.xml=sce_sys/livearea/contents/template.xml
        AnimalCrossingVita.vpk
    DEPENDS AnimalCrossingVita.self-self
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Creating VPK with all LiveArea assets"
)

# Build information
add_custom_target(build_info ALL
    COMMAND echo "🎮 Animal Crossing Vita v${VITA_VERSION}"
    COMMAND echo "Build Type: ${AC_BUILD_TYPE}"
    COMMAND echo "Debug: ${AC_ENABLE_DEBUG}"
    COMMAND echo "AC-Decomp: ${AC_INCLUDE_DECOMP}"
    COMMAND echo "Generated: $$(date)"
    VERBATIM
)
